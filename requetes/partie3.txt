# Exercice 3.2 — match contre term

GET offres/_search
{ "query": { "match": { "description": "projets bancaires" } } }

# 0 résultat : "paris" != "Paris" (keyword = sensible à la casse)
GET offres/_search
{ "query": { "term": { "ville": "paris" } } }

# 0 résultat : titre est un champ text, indexé en tokens ("data", "engineer", "senior")
GET offres/_search
{ "query": { "term": { "titre": "Data Engineer Senior" } } }

# --- Corrections ---

GET offres/_search
{ "query": { "term": { "ville": "Paris" } } }

GET offres/_search
{ "query": { "term": { "titre.brut": "Data Engineer Senior" } } }

# operator and : tous les tokens doivent être présents
GET offres/_search
{
  "query": {
    "match": { "description": { "query": "projets bancaires", "operator": "and" } }
  }
}

# Exercice 3.3 — multi_match, fuzziness, pondération

# a) sans tolérance : "kubernetis" n'existe pas dans l'index

GET offres/_search
{
  "query": {
    "multi_match": {
      "query": "kubernetis terraform",
      "fields": ["titre", "competences.texte", "description"]
    }
  }
}

# b) avec fuzziness

GET offres/_search
{
  "query": {
    "multi_match": {
      "query": "kubernetis terraform",
      "fields": ["titre", "competences.texte", "description"],
      "fuzziness": "AUTO"
    }
  }
}

# c) avec poids sur le titre

GET offres/_search
{
  "query": {
    "multi_match": {
      "query": "kubernetis terraform",
      "fields": ["titre^3", "competences.texte", "description"],
      "fuzziness": "AUTO"
    }
  }
}

# Exercice 3.4 — Requête bool

GET offres/_search
{
  "query": {
    "bool": {
      "must": [
        { "multi_match": { "query": "données", "fields": ["titre", "description"] } }
      ],
      "filter": [
        { "term":  { "contrat": "CDI" } },
        { "terms": { "ville": ["Montpellier", "Toulouse"] } },
        { "range": { "salaire_max": { "gte": 50000 } } }
      ],
      "must_not": [
        { "term": { "teletravail": "aucun" } }
      ],
      "should": [
        { "term": { "competences": "Elasticsearch" } }
      ]
    }
  }
}
